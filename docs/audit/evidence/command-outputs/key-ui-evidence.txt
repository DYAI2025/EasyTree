    11	/**
    12	 * Calculates hours from HH:mm to HH:mm string.
    13	 * Example: "07:00" to "15:30" => 8.5 hours.
    14	 */
    15	export function calculateHours(startTime: string, endTime: string): number {
    16	  if (!startTime || !endTime) return 8; // default 8h
    17	  const [startH, startM] = startTime.split(':').map(Number);
    18	  const [endH, endM] = endTime.split(':').map(Number);
    19	
    20	  const startDecimal = startH + (startM || 0) / 60;
    21	  const endDecimal = endH + (endM || 0) / 60;
    22	
    23	  let diff = endDecimal - startDecimal;
    24	  if (diff <= 0) diff += 24; // boundary fallback
    25	  return Math.max(0, Math.round(diff * 10) / 10);
    26	}

    28	/**
    29	 * Checks time overlap between two time slots on the same day.
    30	 */
    31	export function isTimeOverlapping(
    32	  start1: string,
    33	  end1: string,
    34	  start2: string,
    35	  end2: string
    36	): boolean {
    37	  const parse = (t: string) => {
    38	    const [h, m] = t.split(':').map(Number);
    39	    return h * 60 + (m || 0);
    40	  };
    41	  const s1 = parse(start1);
    42	  const e1 = parse(end1);
    43	  const s2 = parse(start2);
    44	  const e2 = parse(end2);
    45	
    46	  return Math.max(s1, s2) < Math.min(e1, e2);
    47	}

    72	      const matchingAbsence = absences.find(
    73	        (abs) =>
    74	          abs.employeeId === empId &&
    75	          abs.status === 'genehmigt' &&
    76	          asg.date >= abs.startDate &&
    77	          asg.date <= abs.endDate
    78	      );
    79	
    80	      if (matchingAbsence) {
    81	        const emp = employeeMap.get(empId);
    82	        const site = worksiteMap.get(asg.worksiteId);
    83	        const empName = emp ? `${emp.firstName} ${emp.lastName}` : 'Mitarbeiter';
    84	        const siteName = site ? site.name : 'Baustelle';
    85	
    86	        conflicts.push({
    87	          id: `conf-abs-${asg.id}-${empId}`,
    88	          type: 'absence_conflict',
    89	          severity: 'blocking',
    90	          title: 'Mitarbeiter im Urlaub/Abwesend',
    91	          message: `${empName} ist am ${asg.date} (${matchingAbsence.type}) abwesend, aber auf Baustelle "${siteName}" eingeplant.`,
    92	          affectedDate: asg.date,
    93	          affectedEmployeeIds: [empId],
    94	          affectedWorksiteId: asg.worksiteId,
    95	          suggestedFix: `Entferne ${empName} am ${asg.date} oder ersetze die Person.`,
    96	        });
    97	      }
    98	    }
    99	  }
   100	
   101	  // 2. DOUBLE BOOKINGS (Employee scheduled on multiple overlapping assignments)
   102	  const assignmentsByDate = new Map<string, WorksiteAssignment[]>();
   103	  for (const asg of assignments) {
   104	    const list = assignmentsByDate.get(asg.date) || [];
   105	    list.push(asg);
   106	    assignmentsByDate.set(asg.date, list);
   107	  }
   108	
   109	  assignmentsByDate.forEach((dayAssignments, date) => {
   110	    for (let i = 0; i < dayAssignments.length; i++) {
   111	      for (let j = i + 1; j < dayAssignments.length; j++) {
   112	        const a1 = dayAssignments[i];
   113	        const a2 = dayAssignments[j];
   114	
   115	        if (isTimeOverlapping(a1.startTime, a1.endTime, a2.startTime, a2.endTime)) {
   116	          // Check common employees
   117	          const commonEmps = a1.assignedEmployeeIds.filter((id) =>
   118	            a2.assignedEmployeeIds.includes(id)
   119	          );
   120	
   121	          for (const empId of commonEmps) {
   122	            const emp = employeeMap.get(empId);
   123	            const w1 = worksiteMap.get(a1.worksiteId);
   124	            const w2 = worksiteMap.get(a2.worksiteId);
   125	            const empName = emp ? `${emp.firstName} ${emp.lastName}` : 'Mitarbeiter';
   126	
   127	            conflicts.push({
   128	              id: `conf-db-${date}-${empId}-${a1.id}-${a2.id}`,
   129	              type: 'double_booking',
   130	              severity: 'blocking',
   131	              title: 'Doppelbelegung Mitarbeiter',
   132	              message: `${empName} ist am ${date} zeitgleich auf "${w1?.name || 'Baustelle 1'}" (${a1.startTime}-${a1.endTime}) und "${w2?.name || 'Baustelle 2'}" (${a2.startTime}-${a2.endTime}) eingeplant.`,
   133	              affectedDate: date,
   134	              affectedEmployeeIds: [empId],
   135	              affectedWorksiteId: a1.worksiteId,
   136	              suggestedFix: `Verschiebe die Einsatzzeit oder entferne ${empName} von einer Baustelle.`,
   137	            });
   138	          }
   139	
   140	          // Check common EXCLUSIVE equipment
   141	          const commonEquipment = a1.assignedEquipmentIds.filter((id) =>
   142	            a2.assignedEquipmentIds.includes(id)
   143	          );
   144	
   145	          for (const eqId of commonEquipment) {
   146	            const eq = equipmentMap.get(eqId);
   147	            if (eq && eq.isExclusive) {
   148	              const w1 = worksiteMap.get(a1.worksiteId);
   149	              const w2 = worksiteMap.get(a2.worksiteId);
   150	
   151	              conflicts.push({
   152	                id: `conf-eqdb-${date}-${eqId}-${a1.id}-${a2.id}`,
   153	                type: 'resource_double_booking',
   154	                severity: 'blocking',
   155	                title: 'Doppelreservierung Spezialgerät',
   156	                message: `Exklusives Gerät "${eq.name}" ist am ${date} zeitgleich auf "${w1?.name}" und "${w2?.name}" reserviert.`,
   157	                affectedDate: date,
   158	                affectedWorksiteId: a1.worksiteId,
   159	                affectedResourceId: eqId,
   160	                suggestedFix: `Weise ein alternatives Gerät zu oder passe die Einsatzzeiten an.`,
   161	              });
   162	            }
   163	          }
   164	
   165	          // Check common vehicles
   166	          const commonVehicles = a1.assignedVehicleIds.filter((id) =>
   167	            a2.assignedVehicleIds.includes(id)
   168	          );
   169	
   170	          for (const vehId of commonVehicles) {
   171	            const veh = vehicleMap.get(vehId);
   172	            const w1 = worksiteMap.get(a1.worksiteId);
   173	            const w2 = worksiteMap.get(a2.worksiteId);
   174	
   175	            conflicts.push({
   176	              id: `conf-vehdb-${date}-${vehId}-${a1.id}-${a2.id}`,
   177	              type: 'resource_double_booking',
   178	              severity: 'blocking',
   179	              title: 'Doppelreservierung Fahrzeug',
   180	              message: `Fahrzeug "${veh?.name || 'Fahrzeug'}" (${veh?.licensePlate}) ist am ${date} doppelt reserviert für "${w1?.name}" und "${w2?.name}".`,
   181	              affectedDate: date,
   182	              affectedWorksiteId: a1.worksiteId,
   183	              affectedResourceId: vehId,
   184	              suggestedFix: `Ändere die Fahrzeugzuordnung für eine der Baustellen.`,
   185	            });
   186	          }
   187	        }
   188	      }
   189	    }
   190	  });
   191	
   192	  // 3. EMPLOYEE WEEKLY OVERTIME / CAPACITY EXCEEDED
   193	  const employeeWeeklyHours = new Map<string, number>();
   194	  for (const asg of assignments) {
   195	    const duration = calculateHours(asg.startTime, asg.endTime);
   196	    for (const empId of asg.assignedEmployeeIds) {
   197	      const current = employeeWeeklyHours.get(empId) || 0;
   198	      employeeWeeklyHours.set(empId, current + duration);
   199	    }
   200	  }
   201	
   202	  employeeWeeklyHours.forEach((plannedHours, empId) => {
   203	    const emp = employeeMap.get(empId);
   204	    if (emp) {
   205	      if (plannedHours > emp.maxWeeklyHours + 4) {
   206	        // Hard overtime warning
   207	        conflicts.push({
   208	          id: `conf-cap-exceed-${empId}`,
   209	          type: 'capacity_exceeded',
   210	          severity: 'blocking',
   211	          title: 'Kritische Überplanung (>44h)',
   212	          message: `${emp.firstName} ${emp.lastName} ist mit ${plannedHours}h deutlich über der vertraglichen Wochenarbeitszeit (${emp.maxWeeklyHours}h) eingeplant.`,
   213	          affectedDate: 'Ganze Woche',
   214	          affectedEmployeeIds: [empId],
   215	          suggestedFix: `Reduziere die Schichten von ${emp.firstName} ${emp.lastName} um ${plannedHours - emp.maxWeeklyHours} Stunden.`,
   216	        });
   217	      } else if (plannedHours > emp.maxWeeklyHours) {
   218	        // Soft overtime warning
   219	        conflicts.push({
   220	          id: `conf-cap-warn-${empId}`,
   221	          type: 'overtime_warning',
   222	          severity: 'warning',
   223	          title: 'Überstundenwarnung',
   224	          message: `${emp.firstName} ${emp.lastName} ist mit ${plannedHours}h leicht über der Sollzeit von ${emp.maxWeeklyHours}h.`,
   225	          affectedDate: 'Ganze Woche',
   226	          affectedEmployeeIds: [empId],
   227	          suggestedFix: `Prüfe, ob der Überstundenaufbau genehmigt ist.`,
   228	        });
   229	      }
   230	    }
   231	  });
   232	
   233	  // 4. MISSING SKILLS ON WORKSITE
   234	  for (const asg of assignments) {
   235	    const worksite = worksiteMap.get(asg.worksiteId);
   236	    if (worksite && worksite.requiredSkills && worksite.requiredSkills.length > 0) {
   237	      const teamEmployees = asg.assignedEmployeeIds
   238	        .map((id) => employeeMap.get(id))
   239	        .filter((e): e is Employee => Boolean(e));
   240	
   241	      const teamSkills = new Set(teamEmployees.flatMap((e) => e.skills));
   242	
   243	      for (const reqSkill of worksite.requiredSkills) {
   244	        if (!teamSkills.has(reqSkill)) {
   245	          conflicts.push({
   246	            id: `conf-skill-${asg.id}-${reqSkill}`,
   247	            type: 'missing_skill',
   248	            severity: 'warning',
   249	            title: 'Fehlende Fähigkeit im Team',
   250	            message: `Auf Baustelle "${worksite.name}" am ${asg.date} fehlt die geforderte Qualifikation "${reqSkill}".`,
   251	            affectedDate: asg.date,
   252	            affectedWorksiteId: asg.worksiteId,
   253	            suggestedFix: `Weise einen Mitarbeiter mit Qualifikation "${reqSkill}" zu.`,
   254	          });
   255	        }
   256	      }
   257	    }
   258	
   259	    // 5. UNASSIGNED / EMPTY TEAM WARNING
   260	    if (asg.assignedEmployeeIds.length === 0) {
   261	      const worksite = worksiteMap.get(asg.worksiteId);
   262	      conflicts.push({
   263	        id: `conf-unassigned-${asg.id}`,
   264	        type: 'unassigned_team',
   265	        severity: 'warning',
   266	        title: 'Unbesetzte Baustelle',
   267	        message: `Baustelle "${worksite?.name || 'Baustelle'}" hat am ${asg.date} noch keine Mitarbeiter zugewiesen.`,
   268	        affectedDate: asg.date,
   269	        affectedWorksiteId: asg.worksiteId,
   270	        suggestedFix: `Füge der Baustelle ein Team hinzu.`,
   271	      });
   272	    }
   273	
   274	    // 6. WEATHER WARNING ON ASSIGNMENT
   275	    if (asg.weatherWarning) {
   276	      const worksite = worksiteMap.get(asg.worksiteId);
   277	      conflicts.push({
   278	        id: `conf-weather-${asg.id}`,
   279	        type: 'weather_warning',
   280	        severity: 'warning',
   281	        title: 'Wetterwarnung',
   282	        message: `Wetterhinweis für "${worksite?.name}" am ${asg.date}: ${asg.weatherWarning}`,
   283	        affectedDate: asg.date,
   284	        affectedWorksiteId: asg.worksiteId,
   285	        suggestedFix: `Prüfe die Sicherheit bei Baumkletter- und Steigereinsätzen.`,
   286	      });
   287	    }
   288	  }
   289	
   290	  return conflicts;
   291	}

    13	/**
    14	 * Calculates capacity summaries for all employees in a week.
    15	 */
    16	export function getEmployeeCapacitySummaries(
    17	  employees: Employee[],
    18	  assignments: WorksiteAssignment[],
    19	  absences: Absence[],
    20	  weekStartDate: string,
    21	  weekEndDate: string
    22	): EmployeeCapacitySummary[] {
    23	  return employees.map((emp) => {
    24	    // Check if absent during week
    25	    const isAbsentThisWeek = absences.some(
    26	      (abs) =>
    27	        abs.employeeId === emp.id &&
    28	        abs.status === 'genehmigt' &&
    29	        abs.startDate <= weekEndDate &&
    30	        abs.endDate >= weekStartDate
    31	    );
    32	
    33	    // Sum planned hours
    34	    let plannedHours = 0;
    35	    let assignmentsCount = 0;
    36	
    37	    assignments.forEach((asg) => {
    38	      if (
    39	        asg.date >= weekStartDate &&
    40	        asg.date <= weekEndDate &&
    41	        asg.assignedEmployeeIds.includes(emp.id)
    42	      ) {
    43	        plannedHours += calculateHours(asg.startTime, asg.endTime);
    44	        assignmentsCount++;
    45	      }
    46	    });
    47	
    48	    const overtimeHours = Math.max(0, plannedHours - emp.maxWeeklyHours);
    49	    const isOverCapacity = plannedHours > emp.maxWeeklyHours;
    50	
    51	    return {
    52	      employeeId: emp.id,
    53	      plannedHours: Math.round(plannedHours * 10) / 10,
    54	      maxHours: emp.maxWeeklyHours,
    55	      overtimeHours: Math.round(overtimeHours * 10) / 10,
    56	      assignmentsCount,
    57	      isOverCapacity,
    58	      isAbsentThisWeek,
    59	    };
    60	  });
    61	}

    68	  assignments: WorksiteAssignment[]
    69	): { totalPersonHours: number; totalAssignments: number; totalAssignedPeople: number } {

    88	  };

    91	/**
    92	 * Calculates resource utilization for vehicles and equipment across a week.
    93	 */
    94	export function getResourceUtilization(
    95	  vehicles: Vehicle[],
    96	  equipment: Equipment[],
    97	  assignments: WorksiteAssignment[],
    98	  worksites: Worksite[]
    99	): ResourceUtilizationSummary[] {
   100	  const worksiteMap = new Map<string, Worksite>(worksites.map((w) => [w.id, w]));
   101	
   102	  const vehicleSummaries: ResourceUtilizationSummary[] = vehicles.map((veh) => {
   103	    const datesReserved = new Set<string>();
   104	    const worksiteNames = new Set<string>();
   105	    let reservationInstances = 0;
   106	
   107	    assignments.forEach((asg) => {
   108	      if (asg.assignedVehicleIds.includes(veh.id)) {
   109	        datesReserved.add(asg.date);
   110	        reservationInstances++;
   111	        const site = worksiteMap.get(asg.worksiteId);
   112	        if (site) worksiteNames.add(site.name);
   113	      }
   114	    });
   115	
   116	    return {
   117	      resourceId: veh.id,
   118	      name: veh.name,
   119	      category: 'vehicle',
   120	      isExclusive: true,
   121	      totalDaysInWeek: 5,
   122	      reservedDaysCount: datesReserved.size,
   123	      isDoubleBooked: reservationInstances > datesReserved.size,
   124	      assignedWorksites: Array.from(worksiteNames),
   125	    };
   126	  });
   127	
   128	  const equipmentSummaries: ResourceUtilizationSummary[] = equipment.map((eq) => {
   129	    const datesReserved = new Set<string>();
   130	    const worksiteNames = new Set<string>();
   131	    let reservationInstances = 0;
   132	
   133	    assignments.forEach((asg) => {
   134	      if (asg.assignedEquipmentIds.includes(eq.id)) {
   135	        datesReserved.add(asg.date);
   136	        reservationInstances++;
   137	        const site = worksiteMap.get(asg.worksiteId);
   138	        if (site) worksiteNames.add(site.name);
   139	      }
   140	    });
   141	
   142	    return {
   143	      resourceId: eq.id,
   144	      name: eq.name,
   145	      category: 'equipment',
   146	      isExclusive: eq.isExclusive,
   147	      totalDaysInWeek: 5,
   148	      reservedDaysCount: datesReserved.size,
   149	      isDoubleBooked: eq.isExclusive && reservationInstances > datesReserved.size,
   150	      assignedWorksites: Array.from(worksiteNames),
   151	    };
   152	  });
   153	
   154	  return [...vehicleSummaries, ...equipmentSummaries];
   155	}

   160	export interface BentoSummaryMetrics {
   161	  totalEmployees: number;
   162	  plannedEmployeesCount: number;
   163	  unplannedEmployeesCount: number;
   164	  absentEmployeesCount: number;
   165	  activeWorksitesCount: number;
   166	  incompleteWorksitesCount: number;
   167	  totalPlannedHours: number;
   168	  totalCapacityHours: number;
   169	  vehiclesInUseCount: number;
   170	  totalVehicles: number;
   171	  equipmentReservedCount: number;
   172	  totalEquipment: number;
   173	  resourceBottlenecksCount: number;
   174	  blockingConflictsCount: number;
   175	  warningsCount: number;
   176	}
   177	
   178	export function computeBentoMetrics(
   179	  employees: Employee[],
   180	  assignments: WorksiteAssignment[],
   181	  absences: Absence[],
   182	  worksites: Worksite[],
   183	  vehicles: Vehicle[],
   184	  equipment: Equipment[],
   185	  conflicts: any[],
   186	  weekStartDate: string,
   187	  weekEndDate: string
   188	): BentoSummaryMetrics {
   189	  const capacities = getEmployeeCapacitySummaries(
   190	    employees,
   191	    assignments,
   192	    absences,
   193	    weekStartDate,
   194	    weekEndDate
   195	  );
   196	
   197	  const plannedEmployeesCount = capacities.filter((c) => c.plannedHours > 0).length;
   198	  const absentEmployeesCount = capacities.filter((c) => c.isAbsentThisWeek).length;
   199	  const unplannedEmployeesCount = capacities.filter(
   200	    (c) => c.plannedHours === 0 && !c.isAbsentThisWeek
   201	  ).length;
   202	
   203	  const activeWorksiteIds = new Set(
   204	    assignments
   205	      .filter((a) => a.date >= weekStartDate && a.date <= weekEndDate)
   206	      .map((a) => a.worksiteId)
   207	  );
   208	
   209	  const incompleteWorksitesCount = worksites.filter((w) => {
   210	    if (!activeWorksiteIds.has(w.id)) return false;
   211	    const siteAssignments = assignments.filter(
   212	      (a) => a.worksiteId === w.id && a.date >= weekStartDate && a.date <= weekEndDate
   213	    );
   214	    return siteAssignments.some((a) => a.assignedEmployeeIds.length === 0);
   215	  }).length;
   216	
   217	  let totalPlannedHours = 0;
   218	  capacities.forEach((c) => {
   219	    totalPlannedHours += c.plannedHours;
   220	  });
   221	
   222	  const totalCapacityHours = employees.reduce((sum, e) => sum + e.maxWeeklyHours, 0);
   223	
   224	  const utilizations = getResourceUtilization(vehicles, equipment, assignments, worksites);
   225	  const vehiclesInUseCount = utilizations.filter(
   226	    (u) => u.category === 'vehicle' && u.reservedDaysCount > 0
   227	  ).length;
   228	  const equipmentReservedCount = utilizations.filter(
   229	    (u) => u.category === 'equipment' && u.reservedDaysCount > 0
   230	  ).length;
   231	
   232	  const resourceBottlenecksCount = utilizations.filter((u) => u.isDoubleBooked).length;
   233	
   234	  const blockingConflictsCount = conflicts.filter((c) => c.severity === 'blocking').length;
   235	  const warningsCount = conflicts.filter((c) => c.severity === 'warning').length;
   236	
   237	  return {
   238	    totalEmployees: employees.length,
   239	    plannedEmployeesCount,
   240	    unplannedEmployeesCount,
   241	    absentEmployeesCount,
   242	    activeWorksitesCount: activeWorksiteIds.size,
   243	    incompleteWorksitesCount,
   244	    totalPlannedHours: Math.round(totalPlannedHours),
   245	    totalCapacityHours,
   246	    vehiclesInUseCount,
   247	    totalVehicles: vehicles.length,
   248	    equipmentReservedCount,
   249	    totalEquipment: equipment.length,
   250	    resourceBottlenecksCount,
   251	    blockingConflictsCount,
   252	    warningsCount,
   253	  };
   254	}
